def put_table(tabs, name, cols):
    n = 0
    d = {}
    for k, arr in cols.items():
        n = len(arr)
        d[k] = arr
        d[k + "?"] = np.ones(len(arr), dtype=bool)
    tabs.data[name] = d
    tabs.rows[name] = n

def tokenise(tabs):
    """Every text column gets two companions: <name>~h (text before the first "/") and <name>~t (text after it)."""
    for tname, cols in tabs.data.items():
        for name in [c for c in cols if not c.endswith("?")]:
            arr = cols[name]
            if arr.dtype != object:
                continue
            ok = cols[name + "?"]
            n = len(arr)
            h = np.empty(n, dtype=object); t = np.empty(n, dtype=object)
            hm = np.zeros(n, dtype=bool); tm = np.zeros(n, dtype=bool)
            for i in range(n):
                v = arr[i]
                if ok[i] and isinstance(v, str):
                    a, sep, b = v.partition("/")
                    h[i] = a; hm[i] = True
                    if sep:
                        t[i] = b; tm[i] = True
            cols[name + "~h"], cols[name + "~h?"] = h, hm
            cols[name + "~t"], cols[name + "~t?"] = t, tm

def add_product(tabs, name, parent, n_other, col):
    """All pairs (row of parent, 0..n_other-1): the index set of a two-index sum."""
    n = tabs.rows.get(parent, 0)
    put_table(tabs, name, {"_parent": np.repeat(np.arange(n, dtype=float), n_other),
                           col: np.tile(np.arange(n_other, dtype=float), n)})

def add_events(tabs, name, parent, horizon, cycle=None, lists=None):
    """Knot lattices of every parent row: n*cycle (source CYCLE) and the listed times (source LIST), sorted,
    0 < t < horizon*(1-eps), each located in the year grid (cell = index of the year that holds it)."""
    par, tau, src = [], [], []
    n = tabs.rows.get(parent, 0)
    pdata = tabs.data.get(parent, {})
    if cycle and cycle in pdata and pdata[cycle].dtype != object:
        cyc, ok = pdata[cycle], pdata[cycle + "?"]
        for i in range(n):
            if ok[i]:
                for t in rs.periodic_times(horizon, float(cyc[i])):
                    par.append(i); tau.append(t); src.append("CYCLE")
    if lists and lists in tabs.data and tabs.rows[lists]:
        lt = tabs.data[lists]
        by = {}
        for p, v, okv in zip(lt["_parent"], lt["value"], lt["value?"]):
            if okv:
                by.setdefault(int(p), []).append(float(v))
        for i in sorted(by):
            for t in sorted(rs.explicit_times(horizon, by[i])):
                par.append(i); tau.append(t); src.append("LIST")
    s = np.empty(len(src), dtype=object)
    for i, v in enumerate(src):
        s[i] = v
    put_table(tabs, name, {"_parent": np.array(par, dtype=float), "tau": np.array(tau, dtype=float),
                           "cell": np.array([rs.event_year(t) - 1 for t in tau], dtype=float), "src": s})

def strs(xs):
    a = np.empty(len(xs), dtype=object)
    for i, v in enumerate(xs):
        a[i] = v
    return a

def S1(eng, name):
    c = eng.col("S", name)
    assert bool(c.m[0]), ("S", name, "missing")
    return float(c.v[0])
def one(eng, table, name):
    c = eng.col(table, name)

def cv(eng, table, name):
    c = eng.col(table, name)
    assert bool(c.m.all()), (table, name, "missing values")
    return c.v

                    first[y] = tau
            for y, tau in first.items():
                P["B3-01"].append(rs.year_cell(-a, y, gate=tau))
        def counted(name, a, ev):
            if a != 0.0:
                P[name].append(rs.constant(a))
                P[name] += [rs.atom(a, tau) for tau in ev]
        for i, row in enumerate(scn.materials):
            ev = rs.periodic_times(H, row.refresh_cycle)
            self.events["m"].append(ev)
            P["B4-02"] += [rs.atom(float(u_mat[i]), tau) for tau in ev]
            skip_cells(float(a_mat[i]), ev)
            if ev and np_m[i]:
                P["N:b42"].append(rs.atom(1.0, ev[0]))
            counted("X-10r", float(xr[i]), ev); counted("X-10w", float(xw[i]), ev)
        for j, eq in enumerate(scn.equipment):
            ev = rs.explicit_times(H, eq.replacement_event_list) if eq.replacement_event_list is not None else rs.periodic_times(H, eq.service_life)
            self.events["e"].append(ev)
            P["B4-02"] += [rs.atom(float(u_eq[j]), tau) for tau in ev]
            skip_cells(float(a_eq[j]), ev)
            if ev and np_e[j]:
                P["N:b42"].append(rs.atom(1.0, min(ev)))
            P["B4-01n"] += [rs.atom(1.0, tau) for tau in ev]
            counted("X-10w", float(xwe[j]), ev)
        b5u = S1(eng, "b5.u")
        if b5u:
            P["B5-01"] += [rs.atom(b5u, tau) for tau in rs.periodic_times(H, scn.boundary.replacement_cycle)]
        # refrigerants: every amplitude comes from the sum-product columns on the device-year and event tables
        if eng.nrows(DY):
            k, a2, y4, e4a, e4g = (cv(eng, DY, c) for c in ("k", "b1.a2", "b1.y4", "b1.e4a", "b1.e4g"))
            tau1 = eng.col(DY, "b1.tau1").v
            for i in np.nonzero(a2)[0]: P["B1-02"].append(rs.year_cell(float(a2[i]), int(k[i]) + 1))
            for i in np.nonzero(y4)[0]: P["B1-04"].append(rs.year_cell(float(y4[i]), int(k[i]) + 1))
            for i in np.nonzero(e4a)[0]: P["B1-04"].append(rs.atom(float(e4a[i]), float(tau1[i])))
            for i in np.nonzero(e4g)[0]: P["B1-04"].append(rs.year_cell(float(e4g[i]), int(k[i]) + 1, gate=float(tau1[i])))
            P["B1-04"] += [rs.constant(float(c)) for c in cv(eng, D, "b1.k4m0") if c]
        if eng.nrows(E):
            tau, cell, a3, g3 = (cv(eng, E, c) for c in ("tau", "cell", "b1.a3", "b1.g3"))
            for i in np.nonzero(a3)[0]: P["B1-03"].append(rs.atom(float(a3[i]), float(tau[i])))
            for i in np.nonzero(g3)[0]: P["B1-03"].append(rs.year_cell(float(g3[i]), int(cell[i]) + 1, gate=float(tau[i])))
        # operation: year cells with the yearly coefficients of the sum-product; counts of differing years
        kc = cv(eng, CAL, "_pos")
        for name, col in (("B6-03", "b63.a"), ("B2-01", "b21.a"), ("B7-04", "b74.a"), ("X-11", "x11.a")):
            a = cv(eng, CAL, col)
            for i in np.nonzero(a)[0]: P[name].append(rs.year_cell(float(a[i]), int(kc[i]) + 1))
        for name, col in (("N:b64", "b64.mis"), ("N:b62", "b62.mis"), ("N:b71", "p.wat.m.mis"), ("N:b72", "p.sew.m.mis"),
                          ("N:b73", "p.msw.m.mis"), ("N:x06", "x06.m.mis")):
            a = cv(eng, CAL, col)
            for i in np.nonzero(a)[0]: P[name].append(rs.atom(float(a[i]), float(kc[i])))
        return P

    def at(self, T):
        eng, tabs = self.eng, self.tabs
        v = dict(self.const)
        v.update({u: self.K_(u).value(T) for u in T_UNITS})
        self.b14_knot = v["B1-04"]
        if LOCAL_B14:
            put_table(tabs, "TT", {"T": np.array([float(T)])})
            for key in [k for k in eng.cols if k[0] == "TT"]:
                del eng.cols[key]
            eng.run(P2A)
            v["B1-04"] = float(eng.col("S", "b14.loc").v[0]) if eng.nrows(DY) else 0.0
        bad = dict(self.bad_const)
        bad["B4-02"] = self.K_("N:b42").value(T)
        mis = {n: self.K_("N:" + n).value(T) for n in ("b64", "b62", "b71", "b72", "b73", "x06")}
        tt = {"T": np.array([float(T)])}
        tt.update({"mis_" + n: np.array([x]) for n, x in mis.items()})
        if self.base_sum3 is not None:
            tt["base_sum3"] = np.array([self.base_sum3])
        put_table(tabs, "TT", tt)
        put_table(tabs, "U", {"id": strs(U_IDS), "value": np.array([v.get(u, 0.0) for u in U_IDS]),
                              "bad": np.array([bad.get(u, 0.0) for u in U_IDS])})
        for key in [k for k in eng.cols if k[0] in ("TT", "U")]:
            del eng.cols[key]
        eng.run(P2)
        s = lambda n: one(eng, "S", n)
        inc = cv(eng, "STG", "x02.inc")
        bs, full, share = cv(eng, "STG", "x02.bs"), cv(eng, "STG", "x02.full"), cv(eng, "STG", "sum7.share")
        uinc = cv(eng, "U", "x02.uinc")
        out = {"units": v, "bad": bad, "mis": mis,
               "by_stage": {st: float(bs[i]) for i, st in enumerate(STAGES) if inc[i]},
               "by_stage_all": {st: float(full[i]) for i, st in enumerate(STAGES)},
               "complete": bool(s("x02.complete")), "incomplete_units": sorted(u for u, f in zip(U_IDS, uinc) if f),
               "sum1": s("sum1"), "sum2": s("sum2"), "sum3": s("sum3"),
               "sum4": s("sum4") if s("sum4.ok") else None, "sum5": s("sum5") if s("sum4.ok") else None, "sum6": s("sum6"),
               "sum7_defined": bool(s("sum7.def")),
               "sum7": {st: float(share[i]) for i, st in enumerate(STAGES) if inc[i]} if s("sum7.def") else {}}
        has = one(eng, "boundary", "s8.has")
        if not has:
            out["sum8"] = {"status": "NO_BASELINE"}
        elif not s("s8.cmp"):
            out["sum8"] = {"status": "NOT_COMPARABLE", "reasons": [n for n, c in (("AREA", "s8.rA"), ("PERIOD", "s8.rT"), ("STAGES", "s8.rS"),
                                                                                  ("GAS_SCOPE", "s8.rG"), ("FACTOR_VERSION", "s8.rF")) if s(c)]}
        else:
            b = self.base_sum3
            out["sum8"] = {"status": "COMPARED", "baseline": b, "delta": s("s8.delta"), "saving": s("s8.sav") if b > 0.0 else None}
        q = {"violations": [], "warnings": []}
        for name, tab, pre, attr in (("equipment", "equipment", "qa2.e", "replacement_event_list"), ("refrigerants", D, "qa2.r", "change_years")):
            if eng.nrows(tab):
                vv, ww = cv(eng, tab, pre + ".v"), cv(eng, tab, pre + ".w")
                q["violations"] += ["%s[%d].%s" % (name, i, attr) for i in np.nonzero(vv)[0]]
                q["warnings"] += ["%s[%d].%s" % (name, i, attr) for i in np.nonzero(ww)[0]]
        q["ok"] = bool(s("qa2.ok"))
        out["qa2"] = q
        out["dmt"] = {n: bool(s("dmt." + n)) for n in ("b64", "b62", "b71", "b72", "b73", "x06")}
        return out
