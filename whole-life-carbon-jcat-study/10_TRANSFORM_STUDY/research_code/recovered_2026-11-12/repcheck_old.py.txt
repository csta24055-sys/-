def colv(eng, table, name):
    c = eng.col(table, name)
    assert bool(c.m.all()), (table, name, "missing values")
    return c.v

def nonneg(x): return x is not None and x >= 0.0 and not math.isinf(x)
def ratio(x): return x is not None and 0.0 <= x <= 1.0

def assemble(scn, eng, res, op_ref):
    """Pieces per unit. Independent of the reference: B4-02, B3-01, B5-01 (amplitudes from the sum-product engine),
    B1-02/03/04 (straight from inputs). Hybrid: B6-03, B2-01, B7-04 (year values from ONE reference run at the horizon)."""
    Y = math.ceil(H)
    P = {u: [] for u in ("B1-02", "B1-03", "B1-04", "B2-01", "B3-01", "B4-02", "B5-01", "B6-03", "B7-04")}
    u_mat, u_eq = colv(eng, "materials", "b4.u_mat") if scn.materials else [], colv(eng, "equipment", "b4.u_eq") if scn.equipment else []
    a_mat, a_eq = colv(eng, "materials", "b3.a_mat") if scn.materials else [], colv(eng, "equipment", "b3.a_eq") if scn.equipment else []
    events = {"m": [], "e": []}
    def skip_cells(a, ev):
        if a == 0.0:
            return
        P["B3-01"].append(rs.density(a, float(Y)))
        first = {}
        for tau in ev:
            y = rs.event_year(tau)
            if y not in first or tau < first[y]:
                first[y] = tau
        for y, tau in first.items():
            P["B3-01"].append(rs.year_cell(-a, y, gate=tau))
    for i, row in enumerate(scn.materials):
        ev = rs.periodic_times(H, row.refresh_cycle)
        events["m"].append(ev)
        P["B4-02"] += [rs.atom(float(u_mat[i]), tau) for tau in ev]
        skip_cells(float(a_mat[i]), ev)
    for j, eq in enumerate(scn.equipment):
        ev = rs.explicit_times(H, eq.replacement_event_list) if eq.replacement_event_list is not None else rs.periodic_times(H, eq.service_life)
        events["e"].append(ev)
        P["B4-02"] += [rs.atom(float(u_eq[j]), tau) for tau in ev]
        skip_cells(float(a_eq[j]), ev)
    b5u = res["AMP"]["b5_u"]
    if b5u:
        P["B5-01"] += [rs.atom(b5u, tau) for tau in rs.periodic_times(H, scn.boundary.replacement_cycle)]
    for dev in scn.refrigerants:
        if not (nonneg(dev.charge_mass_kg) and nonneg(dev.gwp) and ratio(dev.annual_leak_rate)):
            continue
        ev = sorted(rs.explicit_times(H, dev.change_years)) if dev.change_years is not None else rs.periodic_times(H, dev.service_life)
        by_year = {}
        for tau in ev:
            by_year.setdefault(rs.event_year(tau), []).append(tau)
        m0, r, gwp = dev.charge_mass_kg, dev.annual_leak_rate, dev.gwp
        loss_ok, rec_ok = ratio(dev.replacement_loss_rate), ratio(dev.end_of_life_recovery_rate)
        k4 = (1.0 - dev.end_of_life_recovery_rate) * gwp if rec_ok else 0.0
        if rec_ok:
            P["B1-04"].append(rs.constant(k4 * m0))
        m = m0
        for y in range(1, Y + 1):
            P["B1-02"].append(rs.year_cell(m * r * gwp, y))
            if not dev.top_up and rec_ok:
                P["B1-04"].append(rs.year_cell(-k4 * m * r, y))
            taus = sorted(by_year.get(y, []))
            for n, tau in enumerate(taus):
                if n == 0:
                    if loss_ok:
                        P["B1-03"].append(rs.atom(dev.replacement_loss_rate * gwp * m, tau))
                        if not dev.top_up:
                            P["B1-03"].append(rs.year_cell(-dev.replacement_loss_rate * gwp * m * r, y, gate=tau))
                    if not dev.top_up and rec_ok:
                        P["B1-04"].append(rs.atom(k4 * (m0 - m), tau))
                        P["B1-04"].append(rs.year_cell(k4 * m * r, y, gate=tau))
                elif loss_ok:
                    P["B1-03"].append(rs.atom(dev.replacement_loss_rate * gwp * m0, tau))
            if taus:
                m = m0
            elif not dev.top_up:
                m = m - m * r
    # hybrid part: operation
    elec, fuel = op_ref["B6-01"]["by_year"], op_ref["B6-02"]["by_year"]
    share = scn.operation.facility_management_ratio
    for y in range(1, Y + 1):
        P["B6-03"].append(rs.year_cell(elec[y] + fuel[y], y))
        if ratio(share):
            P["B2-01"].append(rs.year_cell(elec[y] * share, y))
        P["B7-04"].append(rs.year_cell(op_ref["B7-01"]["by_year"][y] + op_ref["B7-02"]["by_year"][y] + op_ref["B7-03"]["by_year"][y], y))
    return P, events, (u_mat, u_eq)

INDEP_UNITS = ("B1-02", "B1-03", "B1-04", "B3-01", "B4-02", "B5-01")
HYBRID_UNITS = ("B6-03", "B2-01", "B7-04")

class Prepared:
    def __init__(self, scn):
        self.scn = scn
        t = time.perf_counter()
        self.tabs = tables.from_scenario(scn)
        self.res, self.eng = msp.evaluate(PROGRAM, self.tabs)
        self.t_msp = time.perf_counter() - t
        top = scenarios.with_period(scn, H)
        self.op_ref = baseline.run_reference(top, ["B6-01", "B6-02", "B7-01", "B7-02", "B7-03"])
        self.P, self.events, self.u = assemble(scn, self.eng, self.res, self.op_ref)
        self.K = {u: KnotSpectrum(p) for u, p in self.P.items()}
        v = lambda uid: self.res[uid]["value"]
        self.const = {"A1-A3": v("A13-01") + v("A13-03") + v("A13-04") + v("X-01"), "A4": v("A4-01") + v("A4-02"), "A5": v("A5-04"),
                      "C1": v("C1-01"), "C2": v("C2-01"), "C3": v("C3-01") + v("X-14"), "C4": v("C4-01")}
        self.n_knots = sum(k.n_knots for k in self.K.values())

    def at(self, T, included):
        v = {u: k.value(T) for u, k in self.K.items()}
        full = dict(self.const)
        full.update({"B1": v["B1-02"] + v["B1-03"] + v["B1-04"], "B2": v["B2-01"], "B3": v["B3-01"], "B4": v["B4-02"],
                     "B5": v["B5-01"], "B6": v["B6-03"] - v["B2-01"], "B7": v["B7-04"]})
        inc = set(included)
        by_stage = {s: full[s] for s in STAGES if s in inc}
        if "B6" in inc and "B2" not in inc:
            by_stage["B6"] = v["B6-03"]
        return v, by_stage, metrics.propagate_stage_change(by_stage, list(inc), {})

STATS = collections.OrderedDict()
ROWS = []
EXCL = []
def record(group, name, cand, ref, case):
    st = STATS.setdefault((group, name), metrics.ErrorStats(name))
    st.add(cand, ref, case)

def run_case(category, label, scn, periods, prep=None):
    prep = prep or Prepared(scn)
    for T in periods:
        cur = scenarios.with_period(scn, T)
        why = scenarios.valid_domain(cur)
        case = "%s|%s|T=%r" % (category, label, T)
        if why:
            EXCL.append((case, why[0]))
            continue
        ref = baseline.run_reference(cur)
        v, by_stage, tot = prep.at(T, cur.boundary.included_lifecycle_stages)
        worst = 0.0
        def cmp(group, name, c, r):
            nonlocal worst
            record(group, name, c, r, case)
            if r != 0.0:
                worst = max(worst, abs(c - r) / abs(r))
            elif c != 0.0:
                worst = max(worst, float("inf") if abs(c) > 1e-6 else 0.0)
        for u in INDEP_UNITS:
            cmp("independent unit", u, v[u], ref[u]["value"])
        for u in HYBRID_UNITS:
            cmp("hybrid unit", u, v[u], ref[u]["value"])
        for s, x in ref["X-02"]["by_stage"].items():
            cmp("independent stage" if s in INDEP_STAGES else "hybrid stage", s, by_stage[s], x)
        cmp("independent total", "SUM-01", tot["SUM-01"], ref["SUM-01"]["value"])
        cmp("hybrid total", "SUM-02", tot["SUM-02"], ref["SUM-02"]["value"])
        cmp("hybrid total", "SUM-03", tot["SUM-03"], ref["SUM-03"]["value"])
        # per-row renewal emission: amplitude x number of events inside T
        rb = ref["B4-02"]
        for i, ev in enumerate(prep.events["m"]):
            n = sum(1 for tau in ev if tau < T * (1.0 - rs.EPS))
            cmp("independent per row", "B4-02 by_item_material", float(prep.u[0][i]) * n, rb["by_item_material"][i])
        for j, ev in enumerate(prep.events["e"]):
            n = sum(1 for tau in ev if tau < T * (1.0 - rs.EPS))
            cmp("independent per row", "B4-02 by_item_equipment", float(prep.u[1][j]) * n, rb["by_item_equipment"][j])
        ROWS.append((category, label, T, worst, len(scn.materials) + len(scn.equipment) + len(scn.refrigerants), prep.n_knots))
    return prep

def cycles_of(scn):
    c = [r.refresh_cycle for r in scn.materials if r.refresh_cycle and r.refresh_cycle > 0]
    c += [e.service_life for e in scn.equipment if e.service_life and e.service_life > 0 and e.replacement_event_list is None]
    return [x for x in c if x <= H]

def scale_factors(scn, k):
    s = copy.deepcopy(scn)
    for f in s.factors.values():
        if f.value is not None:
            f.value *= k
            if f.gas_amounts:
                for g in list(f.gas_amounts):
                    f.gas_amounts[g] *= k
    s.meta.scenario_id += "*f%g" % k
    return s

def scale_quantities(scn, k):
    s = copy.deepcopy(scn)
    for r in s.materials:
        if r.quantity is not None: r.quantity *= k
        if r.transport.mass_t is not None: r.transport.mass_t *= k
        if r.end_of_life.mass_kg is not None: r.end_of_life.mass_kg *= k
    for e in s.equipment:
        if e.quantity is not None: e.quantity *= k
        if e.end_of_life_mass_kg is not None: e.end_of_life_mass_kg *= k
    for d in s.refrigerants:
        if d.charge_mass_kg is not None: d.charge_mass_kg *= k
    s.meta.scenario_id += "*q%g" % k
    return s
